-- SETUP MEDIA FRIENDSHIP EBENHAEZER
-- Jalankan sekali di Supabase SQL Editor.

alter table public.posts add column if not exists media_url text;
alter table public.posts add column if not exists media_type text;

insert into storage.buckets (id,name,public)
values ('friendship-media','friendship-media',true)
on conflict (id) do update set public=true;

drop policy if exists "friendship media upload" on storage.objects;
create policy "friendship media upload"
on storage.objects for insert to authenticated
with check (
  bucket_id='friendship-media'
  and (storage.foldername(name))[1] = auth.uid()::text
);

drop policy if exists "friendship media read" on storage.objects;
create policy "friendship media read"
on storage.objects for select to public
using (bucket_id='friendship-media');

drop policy if exists "friendship media update" on storage.objects;
create policy "friendship media update"
on storage.objects for update to authenticated
using (
  bucket_id='friendship-media'
  and (storage.foldername(name))[1] = auth.uid()::text
)
with check (
  bucket_id='friendship-media'
  and (storage.foldername(name))[1] = auth.uid()::text
);

drop policy if exists "friendship media delete" on storage.objects;
create policy "friendship media delete"
on storage.objects for delete to authenticated
using (
  bucket_id='friendship-media'
  and (storage.foldername(name))[1] = auth.uid()::text
);

-- Pastikan pengguna yang sudah login boleh membuat posting.
drop policy if exists "friendship posts insert" on public.posts;
create policy "friendship posts insert"
on public.posts for insert to authenticated
with check (auth.uid() = user_id);

-- Pengguna dapat melihat posting.
drop policy if exists "friendship posts read" on public.posts;
create policy "friendship posts read"
on public.posts for select to public
using (true);
